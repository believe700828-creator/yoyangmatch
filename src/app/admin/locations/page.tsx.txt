'use client';

import { useState, useEffect } from 'react';

interface LocationItem {
  id: string;
  name: string;
  region: string;
  active: boolean;
}

export default function AdminLocationsPage() {
  const [locations, setLocations] = useState<LocationItem[]>([]);
  const [newName, setNewName] = useState('');
  const [newRegion, setNewRegion] = useState('');

  // 지역 목록 불러오기
  const fetchLocations = async () => {
    try {
      const res = await fetch('/api/locations');
      const data = await res.json();
      if (Array.isArray(data)) {
        setLocations(data);
      }
    } catch (error) {
      console.error('지역 목록을 불러오는 중 오류가 발생했습니다:', error);
    }
  };

  useEffect(() => {
    fetchLocations();
  }, []);

  // 지역 추가 핸들러
  const handleAddLocation = async (e: React.FormEvent) => {
    e.preventDefault();
    if (!newName.trim() || !newRegion.trim()) {
      alert('지역 이름과 권역을 모두 입력해 주세요.');
      return;
    }

    try {
      const res = await fetch('/api/locations', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ name: newName, region: newRegion }),
      });

      if (res.ok) {
        setNewName('');
        setNewRegion('');
        fetchLocations();
      } else {
        alert('지역 추가에 실패했습니다.');
      }
    } catch (error) {
      console.error('지역 추가 오류:', error);
    }
  };

  // 지역 삭제 핸들러
  const handleDeleteLocation = async (id: string) => {
    if (!confirm('정말 이 지역을 삭제하시겠습니까?')) return;

    try {
      const res = await fetch(`/api/locations?id=${id}`, {
        method: 'DELETE',
      });

      if (res.ok) {
        fetchLocations();
      } else {
        alert('지역 삭제에 실패했습니다.');
      }
    } catch (error) {
      console.error('지역 삭제 오류:', error);
    }
  };

  return (
    <div className="max-w-4xl mx-auto p-6">
      <h1 className="text-2xl font-bold mb-6 text-gray-800">📍 요양웹 실시간 지역 관리</h1>

      {/* 지역 추가 폼 */}
      <form onSubmit={handleAddLocation} className="bg-white p-4 rounded-lg shadow mb-6 flex gap-3">
        <input
          type="text"
          placeholder="지역 이름 (예: 수원시 영통구)"
          value={newName}
          onChange={(e) => setNewName(e.target.value)}
          className="flex-1 border p-2 rounded"
        />
        <input
          type="text"
          placeholder="권역/시도 (예: 경기)"
          value={newRegion}
          onChange={(e) => setNewRegion(e.target.value)}
          className="w-40 border p-2 rounded"
        />
        <button type="submit" className="bg-blue-600 text-white px-4 py-2 rounded hover:bg-blue-700">
          지역 추가
        </button>
      </form>

      {/* 지역 목록 테이블 */}
      <div className="bg-white rounded-lg shadow overflow-hidden">
        <table className="w-full text-left border-collapse">
          <thead>
            <tr className="bg-gray-100 border-b">
              <th className="p-3 font-semibold text-gray-700">지역명</th>
              <th className="p-3 font-semibold text-gray-700">권역</th>
              <th className="p-3 font-semibold text-gray-700 text-center">관리</th>
            </tr>
          </thead>
          <tbody>
            {locations.length > 0 ? (
              locations.map((loc) => (
                <tr key={loc.id} className="border-b hover:bg-gray-50">
                  <td className="p-3 text-gray-800">{loc.name}</td>
                  <td className="p-3 text-gray-600">{loc.region}</td>
                  <td className="p-3 text-center">
                    <button
                      onClick={() => handleDeleteLocation(loc.id)}
                      className="bg-red-500 text-white px-3 py-1 rounded text-sm hover:bg-red-600"
                    >
                      삭제
                    </button>
                  </td>
                </tr>
              ))
            ) : (
              <tr>
                <td colSpan={3} className="p-6 text-center text-gray-500">
                  등록된 지역이 없습니다. 새로운 지역을 추가해 보세요!
                </td>
              </tr>
            )}
          </tbody>
        </table>
      </div>
    </div>
  );
}
